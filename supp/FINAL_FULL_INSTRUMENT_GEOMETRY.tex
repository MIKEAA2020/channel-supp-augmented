\begin{theorem}[Centred in-radius]
Assume $nd_B>1$, $s=\min\{d_A,d_B\}$. $\rho_n=2/(nd_Bs)$ in-radius centred at $\mathcal I_*$.
\end{theorem}
\begin{proof}

$\|Z_k\|_\infty\le\|\Tr_B|Z_k|\|_\infty\le s\|Z_k\|_1/d_A$.
If $\|Z\|_1\le2/(nd_Bs)$ then $\|Z\|_\infty\le\lambda_{\min}$ so $J_*+Z\ge0$.
\end{proof}

\begin{theorem}[Full-direction]
Assume $nd_B>1$, $D=D_n$. For $0\le r\le D-1$ exists continuous $h:S^r\to\Inst$ with $\|h(u)-h(-u)\|_{\diamond}\ge2/d_A$, so $\alpha_r\ge1/d_A$ for $r<D$. $d_A=1$ separation $2$, $\alpha_r=1$.
\end{theorem}
\begin{proof}

For $Z\in S(\mathfrak V)$, Jordan $Z=Z_+-Z_-$, $Z_\pm\ge0$, $Z_+Z_-=0$, flag preserved.
$\Tr_{BC_n}Z=0\implies\Tr_{BC_n}Z_+=\Tr_{BC_n}Z_-$.
$t=\Tr Z_+=\Tr Z_->0$ faithful: if $t=0$ then $Z_+=0$, $Z=-Z_-\le0$, $W=-Z\ge0$, $\Tr_{BC_n}W=0$, basis $|b,c\rangle$, $0=\Tr_{BC_n}W=\sum_{b,c}\langle b,c|W|b,c\rangle\ge0$, each $\ge0$, sum zero $\implies$ each zero $\implies W=0$ contradiction.
$Z\mapsto Z_+$ continuous, $t$ continuous, compact sphere $\inf t>0$ bounded away from zero.
$P=Z_+/t$, $Q=Z_-/t$ states orthogonal supports, common marginal $\mu=\Tr_{BC_n}P=\Tr_{BC_n}Q$, density $\mu\le I_{A_0}$.
If $d_A=1$, $J^+=P$, $J^-=Q$.

$J^+=\lambda P+(1-\lambda)\nu\otimes\tau_0$, $J^-=\lambda Q+(1-\lambda)\nu\otimes\tau_0$.
$J^\pm\ge0$, flag, $\Tr_{BC_n}J^\pm=I_{A_0}/d_A$ valid instruments.

Diamond dominates Choi trace via $|\omega_A\rangle$, so $\|h(Z)-h(-Z)\|_{\diamond}\ge2/d_A$.
Continuity because $t$ bounded away from zero.

\end{proof}

\begin{lemma}[Join]
Let $O:S^a\to\Inst$ supported $W_O$, $R:S^b\to\RepInst$ supported $W_R$, $W_O\perp W_R$, antipodal $2$. Then $H:S^{a+b+1}\to\Inst$ antipodal $2$.
\end{lemma}
\begin{proof}
$S^{a+b+1}=\{(x,z):\|x\|^2+\|z\|^2=1\}$, $\lambda=\|x\|^2$, $u=x/\|x\|$ if $x\neq0$, $v=z/\|z\|$ if $z\neq0$.
$H(y)=\lambda O(u)+(1-\lambda)R(v)$ omitting zero weight.
Convex combination instruments.
Continuity at $\lambda=0,1$: $\|\lambda O(u)\|_1=\lambda\to0$ uniformly, similarly $(1-\lambda)R(v)$.
Orthogonal $W_O\perp W_R$ block-diagonal $\Delta H=\lambda\Delta O\oplus(1-\lambda)\Delta R$, norm additive $\|\Delta H\|_1=\lambda\|\Delta O\|_1+(1-\lambda)\|\Delta R\|_1$.
Replacement norm $2$ independent input, choose input witnessing $2$ for $O(u)$, total $2$.
$\ell$-fold: $\lambda_i=\|y_i\|_2^2$, $H(y)=\sum_{i:y_i\neq0}\lambda_i h_i(u_i)$, $\sum\lambda_i=1$, $\|\lambda_i h_i\|_1=\lambda_i\to0$ continuity, orthogonal supports $\|H(y)-H(-y)\|_1=\sum\lambda_i\|h_i(u_i)-h_i(-u_i)\|_1=2$.
\end{proof}

\section{Finite-program evaluation}
\begin{remark}[Role]
Motivational independent of main theorems. Main results do not rely on no-programming. Records exact finite obstruction and divergence of approximate program dimension.
\end{remark}
\begin{theorem}[No-programming]
Exact finite-dimensional universal processor for all unitaries impossible.
\end{theorem}
\begin{proof}
Pure programs orthogonal for distinct unitaries. Mixed $\pi_U=\sum_jq_j|p_j\rangle\langle p_j|$, $q_j>0$, $U|\psi\rangle\langle\psi|U^\dagger=\sum_jq_j\Gamma(|p_j\rangle\langle p_j|\otimes|\psi\rangle\langle\psi|)$. Each summand density, LHS pure extreme point, so each summand equals $U|\psi\rangle\langle\psi|U^\dagger$ for all $j,\psi$. Fixed $j$, $\Lambda_j(X)=\Gamma(|p_j\rangle\langle p_j|\otimes X)$ channel agrees on rank-one projectors spanning $\Herm(E)$, so equal on $\Herm(E)$ real linearity, complex linearity all $\Lin(E)$. Each pure vector in support programs $\mathcal U_U$ exactly. Supports for distinct unitaries orthogonal, finite dimension cannot contain infinitely many orthogonal subspaces.
\end{proof}
\begin{corollary}[Divergence]
If $\Gamma_j$ dimensions $m_j$ approximate all unitaries error $\varepsilon_j\to0$, then $m_j\to\infty$.
\end{corollary}
\begin{proof}
Suppose bounded $m_j\le m_{\max}$, $M\simeq\mathbb C^{m_{\max}}$, isometries $W_j:M_j\to M$, $P_j=W_jW_j^\dagger$, $\tau_j$, $\Lambda_j(\eta)=W_j^\dagger P_j\eta P_jW_j+\Tr[(I-P_j)\eta]\tau_j$ CPTP, $\Lambda_j(W_j\pi W_j^\dagger)=\pi$. Extended $\widetilde\Gamma_j=\Gamma_j\circ(\Lambda_j\otimes\id_E)$ CPTP compact, subsequence chosen independently of $U$, $\widetilde\Gamma_j\to\Gamma$ diamond. Fix $U$, program $\pi'_j(U)$ error $\le\varepsilon_j+1/j$ exists compactness $\Sstate(M_j)$ continuity, embedded $\pi_j(U)=W_j\pi'_j(U)W_j^\dagger\in\Sstate(M)$ compact gives $U$-dependent subsubsequence $\pi_j(U)\to\pi_U$, processors converge same $\Gamma$, continuity evaluation gives $\Gamma(\pi_U\otimes\cdot)=\mathcal U_U$. Single $\Gamma$ evaluates all unitaries contradiction.
\end{proof}
\begin{remark}[Scope]
Forbids exact deterministic universal finite-dimensional. Not forbid approximate, probabilistic, restricted-family, infinite-dimensional, unbounded register, externally supplied classical description. Finite classical register still finite-dimensional quantum program not exempt. Storing Choi as mathematical state not same as evaluating with fixed CPTP processor.
\end{remark}

\begin{lemma}[Bipartite domination]
$\sigma_{RB}\ge0$, $\rho_R=\Tr_B\sigma_{RB}$, $r_R=\rank\rho_R\implies\sigma_{RB}\le\min\{r_R,d_B\}\rho_R\otimes I_B$. Proof: if $\rho_R=0$ trivial else $R_0=\supp\rho_R$, $\rho_R^{-1/2}$ on $R_0$ zero orthocomplement, $Z=(\rho_R^{-1/2}\otimes I_B)\sigma(\rho_R^{-1/2}\otimes I_B)\ge0$, $\Tr_BZ=I_{R_0}$, $Z=\sum|K_\alpha\rangle\!\rangle\langle\!\langle K_\alpha|$, $\sum K_\alpha^\dagger K_\alpha=I_{R_0}$, unit $|v\rangle\in R_0\otimes B$ matrix $V$ rank $b\le\min\{r_R,d_B\}$, $\langle v|Z|v\rangle=\sum|\Tr(V^\dagger K_\alpha)|^2\le b$, hence $Z\le\min\{r_R,d_B\}I$, conjugate back.
\end{lemma}

Testing $|v,k_0\rangle:=|v\rangle\otimes|k_0\rangle$, $S=\supp\Tr_B|v\rangle\langle v|$, $\dim S\le s$.
\end{lemma}
